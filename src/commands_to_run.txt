cd src/log_processing_service
python app.py

# 4. Detection Service (port 8003)
cd src/detection_service
python app.py

# 5. Incident Management Service (port 8004, dashboard: /)
cd src/incident_management
python app.py

# 6. Response Service (port 8005)
cd src/response_service
python app.py
# If port 8005 is busy, run on 8015 (PowerShell): $env:PORT=8015; python app.py

# 7. Notification Service (port 8006)
cd src/notification_service
python app.py

# 8. Actuator Service (port 8007)
cd src/actuator_service
python app.py

# -----------------------------------------------------------
# 2) Supporting/Optional Modules
# -----------------------------------------------------------

# Requires src/ml_service/model.pkl to exist.
cd src/ml_service
python app.py

# 10. Pipeline Demo Service (port 8016, dashboard: /)
